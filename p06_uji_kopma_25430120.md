## Masukan SQL tiap skenario

```sql
USE kopma_120;

-- UJ-01: nota baru dengan anggota aktif -> diterima (AB-02)
INSERT INTO penjualan (no_nota_penjualan, tgl_penjualan, id_petugas, id_anggota, bayar_penjualan)
SELECT 'PJ-2609-0150', '2026-09-25 09:00:00', p.id_petugas, a.id_anggota, 10000
FROM petugas p, anggota a WHERE p.kode_petugas='K01' AND a.no_anggota='A-0002';

-- UJ-02: nota tanpa anggota (pembeli umum) -> diterima, id_anggota NULL (AB-02)
INSERT INTO penjualan (no_nota_penjualan, tgl_penjualan, id_petugas, id_anggota, bayar_penjualan)
SELECT 'PJ-2609-0151', '2026-09-25 09:30:00', p.id_petugas, NULL, 5000
FROM petugas p WHERE p.kode_petugas='K01';

-- UJ-03: stok diubah -5 -> ditolak ERROR 4025 (AB-03)
UPDATE barang SET stok_barang = -5 WHERE kode_barang = 'ATK-001';

-- UJ-04: nomor nota sudah ada -> ditolak ERROR 1062 (AB-01)
INSERT INTO penjualan (no_nota_penjualan, tgl_penjualan, id_petugas, id_anggota, bayar_penjualan)
SELECT 'PJ-2609-0142', NOW(), p.id_petugas, NULL, 1000 FROM petugas p WHERE p.kode_petugas='K01';

-- UJ-05: detail dengan qty 0 -> ditolak ERROR 4025 (ck_detail_qty)
INSERT INTO detail_penjualan (id_penjualan, id_barang, qty_detail_penjualan, harga_satuan_detail_penjualan)
SELECT j.id_penjualan, b.id_barang, 0, b.harga_jual_barang
FROM penjualan j, barang b WHERE j.no_nota_penjualan='PJ-2609-0150' AND b.kode_barang='ATK-001';

-- UJ-06: anggota dengan NIM kembar -> ditolak ERROR 1062 (AB-05)
INSERT INTO anggota (no_anggota, nim_anggota, nama_anggota, prodi_anggota, status_anggota, tgl_daftar_anggota)
VALUES ('A-0099','2301010101','Uji Kembar','Ilmu Komputer','aktif','2026-09-25');

-- UJ-07: penjualan oleh petugas tak dikenal -> ditolak ERROR 1452 (fk_penjualan_petugas)
INSERT INTO penjualan (no_nota_penjualan, tgl_penjualan, id_petugas, id_anggota, bayar_penjualan)
VALUES ('PJ-2609-0199', NOW(), 99, NULL, 5000);

-- UJ-08: UPDATE tanpa kunci di mode aman -> ditolak ERROR 1175 (kebijakan lab)
SET SESSION sql_safe_updates = 1;
UPDATE barang SET harga_jual_barang = harga_jual_barang * 1.1;

-- UJ-09 (langkah D.3 no.6): hapus nota uji -> detail ikut terhapus (ON DELETE CASCADE)
--   Tambahkan dulu satu baris detail untuk PJ-2609-0150, hitung barisnya, DELETE nota itu
--   (WHERE no_nota_penjualan = ...), hitung lagi baris detailnya.
-- UJ-10 (langkah D.3 no.6): hapus petugas yang pernah melayani nota -> ditolak ERROR 1451
DELETE FROM petugas WHERE kode_petugas = 'K03';
```
